Documentación de Arquitectura de Datos

Diseño de Base de Datos

Plataforma Académica Distribuida para Gestión Escolar y Evaluación por Competencias MINEDU.

Motor: Microsoft SQL Server 2022 Patrón: Database-per-Service Backend: Java 17 + Spring Boot 3.x Frontend: Angular 21 (PWA)
1. Arquitectura Distribuida y Esquema Global 5 Bases de Datos Lógicas

Cada microservicio administra su propio esquema en Microsoft SQL Server 2022. La integridad inter-servicio se preserva mediante identificadores cruzados (Soft Foreign Keys) validados en la capa de servicios vía REST y Spring Cloud Gateway.

    ESTUDIANTES {
        bigint id PK
        string dni UK
        string nombre
        string apellido
        string email UK
        bigint grado_id "Soft FK"
        bigint seccion_id "Soft FK"
        string estado
    }

    USUARIOS {
        bigint id PK
        string username UK
        string password
        string rol
        bigint estudiante_id FK
    }

    APODERADOS {
        bigint id PK
        string dni UK
        string nombre
        string apellido
        string telefono
    }

    APODERADO_ESTUDIANTE {
        bigint id PK
        bigint estudiante_id FK
        bigint apoderado_id FK
        string parentesco
        bit permiso_retiro
    }

    FICHAS_MEDICAS {
        bigint id PK
        bigint estudiante_id FK "UK"
        string tipo_sangre
        string alergias
        string tipo_seguro
    }

    DOCUMENTOS_ADJUNTOS {
        bigint id PK
        bigint estudiante_id FK
        string nombre_archivo
        string tipo_documento
        varbinary datos
    }

    DERIVACIONES_PSICOLOGICAS {
        bigint id PK
        bigint estudiante_id FK
        bigint docente_derivador_id
        string motivo
        string prioridad
        string estado
    }

    NIVELES {
        bigint id PK
        string nombre UK
        bit activo
    }

    GRADOS {
        bigint id PK
        string nombre
        bigint nivel_id FK
        bit activo
    }

    SECCIONES {
        bigint id PK
        string nombre
        bigint grado_id FK
        string turno
        int capacidad_maxima
    }

    DOCENTES {
        bigint id PK
        string dni UK
        string nombre
        string apellido
        string especialidad
    }

    CURSOS {
        bigint id PK
        string codigo UK
        string nombre
        int creditos
        bigint grado_id
    }

    HORARIOS {
        bigint id PK
        string dia_semana
        time hora_inicio
        time hora_fin
        bigint aula_id
        bigint seccion_id
        bigint curso_id FK
        bigint docente_id FK
    }

    NOTAS {
        bigint id PK
        bigint estudiante_id "Soft FK"
        bigint curso_id "Soft FK"
        bigint docente_id "Soft FK"
        string competencia
        string escala_literal
        decimal valor_numerico
    }

    ASISTENCIAS {
        bigint id PK
        date fecha
        string estado
        bigint estudiante_id "Soft FK"
        bigint horario_id "Soft FK"
        bigint docente_id "Soft FK"
    }

    REPORTES_DISCIPLINARIOS {
        bigint id PK
        bigint estudiante_id "Soft FK"
        bigint docente_id "Soft FK"
        string tipo
        string estado
        bit apoderado_notificado
    }
